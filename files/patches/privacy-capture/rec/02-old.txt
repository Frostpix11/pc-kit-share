start_piece() {
  local file="$1"
  if [[ $GSR == 1 ]]; then
    # the selector's "X,Y WxH" as gpu-screen-recorder's WxH+X+Y
    local x y w h
    read -r x y w h < <(tr ',x' '  ' <"$GEOM")
    gpu-screen-recorder -w "${w}x${h}+${x}+${y}" -f "$FPS" -c mp4 "${GENC[@]}" -ac aac -q high -v no "${GAUD[@]}" -o "$file" \
      >>/tmp/niri-record-region.log 2>&1 &
  else
    wf-recorder -g "$(<"$GEOM")" -f "$file" -r "$FPS" -D "${ENC[@]}" "${AUD[@]}" \
      >>/tmp/niri-record-region.log 2>&1 &
  fi
  echo $! >"$PIDFILE"
  disown || true
  echo "$file" >>"$PARTS"
}
